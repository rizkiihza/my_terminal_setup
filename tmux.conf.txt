# ==========================================
# Tmux Configuration
# ==========================================

# ------------------------------------------
# Prefix Key: Ctrl-j
# ------------------------------------------
set-option -g prefix C-j
unbind-key C-b
bind-key C-j send-prefix

# Easy config reload
bind-key R source-file ~/.tmux.conf \; display-message "tmux.conf reloaded."

# ------------------------------------------
# General & Performance
# ------------------------------------------
# Faster command sequences (removes the 500ms Esc delay for Vim)
set -s escape-time 10

# Increase scrollback buffer size
set -g history-limit 50000

# Enable focus events for modern editors (Vim/Neovim)
set -g focus-events on

# Vi mode for navigation and copying
setw -g mode-keys vi

# Terminal colors & TrueColor (24-bit RGB) support
set -g default-terminal "tmux-256color"
set -as terminal-features ",xterm-256color:RGB"
set -ga terminal-overrides ",*256col*:Tc"

# Enable unified mouse support (clicking, resizing, scrolling)
set -g mouse on

# Allow arrow keys immediately after changing windows
set-option -g repeat-time 500

# ------------------------------------------
# Window & Pane Indexing
# ------------------------------------------
# Start windows and panes at 1 (easier to reach on keyboard)
set -g base-index 1
setw -g pane-base-index 1

# Automatically renumber windows when one is closed
set -g renumber-windows on

# ------------------------------------------
# Splits & Windows (Preserve Current Path)
# ------------------------------------------
bind-key v split-window -h -c "#{pane_current_path}"
bind-key s split-window -v -c "#{pane_current_path}"
bind-key c new-window -c "#{pane_current_path}"

# Restore session tree view (since 's' is remapped to split)
bind-key S choose-tree -sZ

# ------------------------------------------
# Pane Navigation & Resizing (Vim-style)
# ------------------------------------------
# Prefix + hjkl pane navigation (repeatable)
bind-key -r h select-pane -L
bind-key -r j select-pane -D
bind-key -r k select-pane -U
bind-key -r l select-pane -R
bind-key a last-pane

# Alt + hjkl for instant pane navigation without prefix
bind -n M-h select-pane -L
bind -n M-j select-pane -D
bind -n M-k select-pane -U
bind -n M-l select-pane -R

# Prefix + Shift-HJKL pane resizing (repeatable)
bind-key -r H resize-pane -L 5
bind-key -r J resize-pane -D 5
bind-key -r K resize-pane -U 5
bind-key -r L resize-pane -R 5

# Window cycling & management
bind-key space next-window
bind-key bspace previous-window
bind-key q next-window
bind-key e previous-window
bind-key enter next-layout
bind-key C-o rotate-window

# Layout presets
bind-key + select-layout main-horizontal
bind-key = select-layout main-vertical
set-window-option -g other-pane-height 25
set-window-option -g other-pane-width 80

# Utilities
bind-key : command-prompt
bind-key r refresh-client
bind-key L clear-history
set-window-option -g display-panes-time 1500

# ------------------------------------------
# Vi Copy Mode & System Clipboard (Wayland / wl-copy)
# ------------------------------------------
bind-key [ copy-mode
bind-key ] paste-buffer

set -s set-clipboard on
bind-key -T copy-mode-vi v send-keys -X begin-selection
bind-key -T copy-mode-vi y send-keys -X copy-pipe-and-cancel "wl-copy"
bind-key -T copy-mode-vi Enter send-keys -X copy-pipe-and-cancel "wl-copy"
bind-key -T copy-mode-vi r send-keys -X rectangle-toggle

# ------------------------------------------
# Status Bar & Visual Polish (Tmux 3.x syntax)
# ------------------------------------------
set-option -g status-interval 2
set-option -g status-style bg=colour235,fg=colour136

# Left status: show session name with distinct styling
set-option -g status-left-length 30
set-option -g status-left '#[fg=colour235,bg=colour136,bold] #S #[bg=default] '

# Right status: show date and time
set-option -g status-right-length 50
set-option -g status-right '#[fg=colour244]%Y-%m-%d #[fg=colour250,bold]%H:%M #[default]'

# Inactive window tabs
setw -g window-status-style fg=colour244,bg=default
setw -g window-status-format ' #I:#W#F '

# Active window tab (highlighted)
setw -g window-status-current-style fg=colour235,bg=colour39,bold
setw -g window-status-current-format ' #I:#W#F '

# Pane border styling (modern syntax)
set -g pane-border-style fg=colour238
set -g pane-active-border-style fg=colour39

# Activity alerts
setw -g monitor-activity on
set -g visual-activity on
