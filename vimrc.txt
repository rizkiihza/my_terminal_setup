" Delete buffer without closing the split window
function! s:BufDelete() abort
  let l:current = bufnr('%')
  let l:alt = bufnr('#')
  if buflisted(l:alt)
    buffer #
  else
    bnext
  endif
  if bufnr('%') != l:current
    execute 'silent bdelete' l:current
  endif
endfunction
command! Bd call s:BufDelete()
nnoremap <silent> <leader>bd :call <SID>BufDelete()<CR>

" Strip trailing whitespace (preserves search history and cursor position)
function! StripWhitespace() abort
  let l:save_cursor = getpos(".")
  let l:old_query = getreg('/')
  keeppatterns %s/\s\+$//e
  call setpos('.', l:save_cursor)
  call setreg('/', l:old_query)
endfunction
nnoremap <silent> <leader>ss :call StripWhitespace()<CR>

" Toggle expandtab (tabs vs spaces)
function! ToggleExpandTab() abort
  if &expandtab
    set noexpandtab softtabstop=0
    echo "Indent: Tabs"
  else
    let &softtabstop = &tabstop
    set expandtab
    echo "Indent: Spaces"
  endif
endfunction
nnoremap <silent> <leader>et :call ToggleExpandTab()<CR>
